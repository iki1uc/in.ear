// ===== NC.OS ROOT · ZENTRALER STATE ===== const STATE = { version: "NC.OS ROOT", tree: { id: "root", label: "NC.OS ROOT", level: 0, children: [] }, path: [], // Array von 1-basierten Indizes, z.B. [1, 2] = root→kind[1]→kind[2] currentId: "root", level: 0, seek: 0, log: [], // Append-Only-Log editorText: "", nextId: 1 }; // ===== FIX 1: ZENTRALER STATE — nodeFromPath ===== function nodeFromPath(path) { if (!path || path.length === 0) return STATE.tree; let node = STATE.tree; for (const idx of path) { if (!node.children || !Array.isArray(node.children) || idx < 0 || idx >= node.children.length) { return null; // Pfad ungültig → null zurückgeben } node = node.children[idx]; } return node; } // ===== FIX 2: SYNC — aktuellen Node aus Pfad ableiten ===== function sync() { const n = nodeFromPath(STATE.path); if (!n) { // Pfad ungültig → zurücksetzen auf Root STATE.path = []; STATE.currentId = STATE.tree.id; STATE.level = 0; return; } STATE.currentId = n.id; STATE.level = n.level; // Editor-Inhalt synchronisieren, falls Node vorhanden const editorEl = document.getElementById("editorText"); if (editorEl && n.content !== undefined) { editorEl.value = n.content; STATE.editorText = n.content; } } // ===== SAVE — localStorage Persistenz ===== function save() { try { const { tree, nextId, ...rest } = STATE; // Baum und nächste ID separat speichern const payload = { version: STATE.version, tree: tree, nextId: nextId, path: STATE.path, currentId: STATE.currentId, level: STATE.level, seek: STATE.seek, log: STATE.log.slice(-200), // Letzte 200 Einträge behalten editorText: STATE.editorText }; localStorage.setItem("NC_OS_ROOT", JSON.stringify(payload)); } catch (e) { console.warn("NC.OS ROOT: save fehlgeschlagen", e); } } // ===== RESTORE — localStorage wiederherstellen ===== function restore() { try { const raw = localStorage.getItem("NC_OS_ROOT"); if (!raw) return; const s = JSON.parse(raw); // Validierung: Version und Pfad prüfen if (s && s.version === "NC.OS ROOT" && Array.isArray(s.path)) { STATE.path = s.path; STATE.currentId = s.currentId || STATE.tree.id; STATE.level = s.level || 0; STATE.seek = s.seek || 0; STATE.log = Array.isArray(s.log) ? s.log : []; STATE.editorText = s.editorText || ""; STATE.nextId = s.nextId || 1; if (s.tree) STATE.tree = s.tree; } } catch (e) { console.warn("NC.OS ROOT: restore fehlgeschlagen, localStorage gelöscht"); localStorage.removeItem("NC_OS_ROOT"); } } // ===== FIX 3: APPEND-ONLY LOG — nie löschen, nur anhängen ===== function appendLog(entry) { const t = document.getElementById("editorText"); const text = t ? t.value : ""; if (!text && !entry) return; STATE.seek++; STATE.log.push({ id: STATE.seek, timestamp: Date.now(), nodeId: STATE.currentId, path: [...STATE.path], text: text || (entry && entry.text ? entry.text : ""), action: entry && entry.action ? entry.action : "append" }); // Log-Display aktualisieren ( Append-Only: nie entries entfernen) const logEl = document.getElementById("logDisplay"); if (logEl) { const line = document.createElement("div"); line.className = "log-line"; line.textContent = `[${STATE.seek}] ${text.slice(0, 120)}`; logEl.appendChild(line); logEl.scrollTop = logEl.scrollHeight; // Max 500 Einträge im DOM behalten (Performanz) while (logEl.children.length > 500) { logEl.removeChild(logEl.firstChild); } } // Persistieren save(); } // ===== INIT ===== restore(); sync();
